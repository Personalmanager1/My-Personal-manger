🔐 Personal Manager — Supabase

پہلے Supabase Project URL اور Publishable/Anon Key درج کریں۔ پھر Admin اکاؤنٹ سے Sign in کریں۔ یہ credentials کسی password کی طرح چھپانے کے لیے نہیں ہیں؛ Supabase کا client key ویب ایپ میں public ہوتا ہے۔ Service Role Key یہاں کبھی نہ ڈالیں۔

Supabase میں پہلے SQL table/policies بنائیں، پھر یہاں Sign in کریں۔

Supabase SQL

create table if not exists public.pm_state (
  id text primary key default 'main',
  owner_id uuid not null references auth.users(id) on delete cascade,
  data jsonb not null default '{}'::jsonb,
  updated_at timestamptz not null default now(),
  unique(owner_id)
);
alter table public.pm_state enable row level security;
drop policy if exists "owner select" on public.pm_state;
drop policy if exists "owner insert" on public.pm_state;
drop policy if exists "owner update" on public.pm_state;
drop policy if exists "owner delete" on public.pm_state;
create policy "owner select" on public.pm_state for select using (auth.uid() = owner_id);
create policy "owner insert" on public.pm_state for insert with check (auth.uid() = owner_id);
create policy "owner update" on public.pm_state for update using (auth.uid() = owner_id) with check (auth.uid() = owner_id);
create policy "owner delete" on public.pm_state for delete using (auth.uid() = owner_id);

Supabase Dashboard → SQL Editor میں یہ SQL Run کریں۔ پھر Authentication → Users میں اپنا Admin user بنائیں، اور اسی email/password سے اوپر login کریں۔
